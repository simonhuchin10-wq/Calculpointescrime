function fail(t){err.textContent=t;err.classList.add("show");res.classList.remove("show")}
function clearErr(){err.classList.remove("show")}

typeEl.addEventListener("change",()=>{{
  clearErr();res.classList.remove("show");
  const t=typeEl.value;
  rBox.classList.toggle("hidden",!t);
  pBox.classList.toggle("hidden",!t);
  if(t==="national"){{
    document.getElementById("participantsHint").textContent="L'effectif sert à vérifier que ton classement est cohérent.";

    document.getElementById("participantsHint").textContent="L'effectif détermine automatiquement la tranche de la grille.";
  }}
}});

function calculate(){{
  clearErr();
  const t=typeEl.value, n=Number(pEl.value), rank=Number(rEl.value);

  if(!t)return fail("Sélectionne le type de compétition.");
  if(!Number.isInteger(n)||n<1||n>512)return fail("Le nombre de participants doit être un entier entre 1 et 512.");
  if(!Number.isInteger(rank)||rank<1||rank>512)return fail("Le classement doit être un entier entre 1 et 512.");
  if(rank>n)return fail("Le classement ne peut pas être supérieur au nombre de participants.");

  let value=null, info="";

  if(t==="national"){{
    value=NATIONAL[String(rank)]?.[0] ?? null;
    if(value===null)return fail("Ce classement n'est pas présent dans la grille nationale.");
    badge.textContent="ÉPREUVE NATIONALE";
    info=`${{n}} participants • classement ${{rank}}`;
  }}else{{
    const col=brackets.findIndex(x=>n>=x[0]&&n<=x[1]);
    value=REGIONAL[String(rank)]?.[col] ?? null;
    if(value===null)return fail("Cette case est vide dans la grille : aucun point n'est indiqué pour cette combinaison.");
    badge.textContent="ÉPREUVE RÉGIONALE";
    info=`${{n}} participants • tranche ${{brackets[col][2]}} • classement ${{rank}}`;
  }}

  pts.textContent=value.toLocaleString("fr-FR")+" points";
  details.textContent=info;
  res.classList.add("show");
}}
